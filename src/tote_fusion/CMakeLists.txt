cmake_minimum_required(VERSION 3.16)
project(tote_fusion)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(OpenSSL REQUIRED)
add_library(tote_core src/core.cpp src/geometry.cpp)
target_include_directories(tote_core PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(tote_core PUBLIC Eigen3::Eigen yaml-cpp OpenSSL::Crypto)
foreach(dep rclcpp rclcpp_action tf2 tf2_msgs tf2_ros tf2_eigen sensor_msgs geometry_msgs diagnostic_msgs tote_fusion_interfaces)
  find_package(${dep} REQUIRED)
endforeach()
add_executable(tote_fusion_node src/node.cpp)
target_link_libraries(tote_fusion_node tote_core)
ament_target_dependencies(tote_fusion_node rclcpp rclcpp_action tf2 tf2_msgs tf2_ros tf2_eigen sensor_msgs geometry_msgs diagnostic_msgs tote_fusion_interfaces)
install(TARGETS tote_fusion_node DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_core test/test_core.cpp test/test_geometry.cpp)
  target_link_libraries(test_core tote_core)
  target_compile_definitions(test_core PRIVATE TOTE_MODEL_FILE="${CMAKE_CURRENT_SOURCE_DIR}/config/tote_model.yaml")
  add_executable(evaluate_noise test/evaluate_noise.cpp)
  target_link_libraries(evaluate_noise tote_core)
  target_compile_definitions(evaluate_noise PRIVATE TOTE_MODEL_FILE="${CMAKE_CURRENT_SOURCE_DIR}/config/tote_model.yaml")
  add_executable(evaluate_geometry test/evaluate_geometry.cpp)
  target_link_libraries(evaluate_geometry tote_core)
  target_compile_definitions(evaluate_geometry PRIVATE TOTE_MODEL_FILE="${CMAKE_CURRENT_SOURCE_DIR}/config/tote_model.yaml")
  find_package(launch_testing_ament_cmake REQUIRED)
  add_launch_test(test/test_action.py TIMEOUT 90)
  add_launch_test(test/test_multiview.py TIMEOUT 90)
  add_launch_test(test/test_geometry_action.py TIMEOUT 90)
endif()
install(TARGETS tote_core EXPORT tote_core ARCHIVE DESTINATION lib LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
ament_export_targets(tote_core HAS_LIBRARY_TARGET)
ament_export_dependencies(Eigen3 yaml-cpp OpenSSL)
ament_package()
